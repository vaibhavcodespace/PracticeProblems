Article Sorting — HackerRank Problem Statement

You need to create a React application called Article Sorting that displays a list of articles and allows the user to sort them in two different ways. Some of the application is already implemented, and your task is to complete the remaining functionality.


import "h8k-components";

import Articles from "./components/Articles";
import {useState} from 'react';
import "./App.css";

function App({ articles }) {
  const [isSortedBy,setIsSortedBy] = useState('upvotes')
  const handleMostUpvoted = () => {
    // Logic for most upvoted articles
    setIsSortedBy('upvotes')
  };

  const handleMostRecent = () => {
    // Logic for most recent articles
    setIsSortedBy('recent')
  };
  return (
    <>
      <h8k-navbar header="Sorting Articles"></h8k-navbar>
      <div className="App">
        <div className="layout-row align-items-center justify-content-center my-20 navigation">
          <label className="form-hint mb-0 text-uppercase font-weight-light">
            Sort By
          </label>
          <button
            data-testid="most-upvoted-link"
            className="small"
            onClick={handleMostUpvoted}
          >
            Most Upvoted
          </button>
          <button
            data-testid="most-recent-link"
            className="small"
            onClick={handleMostRecent}
          >
            Most Recent
          </button>
        </div>
        <Articles articles={articles} isSortedBy={isSortedBy} />
      </div>
    </>
  );
}

export default App;



import React from "react";

function Articles({ articles = [], isSortedBy }) {
  let condition = ''
  if (isSortedBy === 'upvotes') {
    condition = articles.sort((a, b) => b.upvotes - a.upvotes)
  } else if (isSortedBy === 'recent') {
    condition = articles.sort((a, b) => new Date(b.date) - new Date(a.date))
  }
  return (
    <div className="card w-50 mx-auto">
      <table>
        <thead>
          <tr>
            <th>Title</th>
            <th>Upvotes</th>
            <th>Date</th>
          </tr>
        </thead>
        <tbody>
          {condition.map((article, index) => (
            <tr data-testid="article" key={`article-${index}`}>
              <td data-testid="article-title">{article.title}</td>
              <td data-testid="article-upvotes">{article.upvotes}</td>
              <td data-testid="article-date">{article.date}</td>
            </tr>
          ))}

        </tbody>
      </table>
    </div>
  );
}

export default Articles;
